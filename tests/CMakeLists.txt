add_executable(uvnet_test_loop test_loop.cpp)
add_executable(uvnet_test_tcp test_tcp.cpp)
add_executable(uvnet_test_udp test_udp.cpp)
add_executable(uvnet_test_dns test_dns.cpp)

foreach(test_name IN ITEMS uvnet_test_loop uvnet_test_tcp uvnet_test_udp uvnet_test_dns)
  target_link_libraries(${test_name} PRIVATE uvnet)
  target_include_directories(${test_name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  if(MSVC)
    target_compile_options(${test_name} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${test_name} PRIVATE -Wall -Wextra)
  endif()
  add_test(NAME ${test_name} COMMAND ${test_name})
  set_tests_properties(${test_name} PROPERTIES TIMEOUT 90)
endforeach()
